# In-cluster Git server for the GitOps mini project (replaces a GitHub repo).
# git daemon serves the bare repo on port 9418:
#  - the developer pushes to it through `kubectl port-forward` (git://localhost:9418/gitops.git)
#  - Argo CD pulls from it inside the cluster (git://git-server.gitserver.svc.cluster.local/gitops.git)
apiVersion: v1
kind: Namespace
metadata:
  name: gitserver
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: git-server
  namespace: gitserver
spec:
  replicas: 1
  selector:
    matchLabels:
      app: git-server
  template:
    metadata:
      labels:
        app: git-server
    spec:
      initContainers:
        - name: init-repo
          image: alpine/git:2.47.2
          command: ["sh", "-c"]
          args:
            - git init --bare --initial-branch=main /srv/git/gitops.git &&
