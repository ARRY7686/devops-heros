# Declarative Pipeline Architecture Specification
pipeline:
  name: Enterprise Production Pipeline
  trigger:
    branches:
      - main
      - develop
    events:
      - push
      - pull_request

  stages:
    - name: source
      type: sequential
      tasks:
        - checkout: git-scm

    - name: build
      type: sequential
      depends_on: [source]
      tasks:
        - compile: python -m compileall src/

    - name: verification
      type: parallel
      depends_on: [build]
      tasks:
        - unit_tests: pytest tests/unit/
        - lint: flake8 src/ tests/
        - security_scan: bandit -r src/

    - name: package
      type: sequential
      depends_on: [verification]
      tasks:
        - build_docker: docker build -t enterprise-app:latest .

    - name: deploy
      type: sequential
      depends_on: [package]
      environment: staging
      tasks:
        - apply_manifests: kubectl apply -f k8s/
